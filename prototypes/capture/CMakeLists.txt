cmake_minimum_required(VERSION 3.21)
project(dome-capture VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
pkg_check_modules(CAPTURE REQUIRED IMPORTED_TARGET libavcodec>=62 libavformat>=62 libavutil>=60 libavfilter>=11
  libpipewire-0.3 gio-unix-2.0 libcurl egl glesv2 gbm libdrm fontconfig zlib)
add_library(capture-core STATIC encoder.cpp sources.cpp eye.cpp audio.cpp third_party/obsmirror/frame_ipc.cpp)
target_include_directories(capture-core PUBLIC . ../../native)
target_link_libraries(capture-core PUBLIC PkgConfig::CAPTURE Threads::Threads)
target_compile_options(capture-core PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
add_executable(dome-capture main.cpp)
target_link_libraries(dome-capture PRIVATE capture-core fltk::fltk-shared fltk::gl-shared fltk::images-shared)
include(../../native/tray/desktop-menu.cmake)
de_xr_desktop_menu(dome-capture)
set_target_properties(dome-capture PROPERTIES INSTALL_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>")
include(CTest)
if(BUILD_TESTING)
  add_executable(capture-test test.cpp)
  target_link_libraries(capture-test PRIVATE capture-core)
  target_compile_options(capture-test PRIVATE -UNDEBUG)
  add_executable(capture-gpu-test gpu-test.cpp)
  target_link_libraries(capture-gpu-test PRIVATE capture-core)
  target_compile_options(capture-gpu-test PRIVATE -UNDEBUG)
  add_executable(capture-client client.cpp)
  target_link_libraries(capture-client PRIVATE fltk::fltk-shared PkgConfig::CAPTURE)
  add_test(NAME capture-core COMMAND capture-test)
  set_tests_properties(capture-core PROPERTIES TIMEOUT 60)
endif()
install(TARGETS dome-capture RUNTIME DESTINATION bin)
install(FILES dome-capture.desktop DESTINATION share/applications)
install(FILES README.md third_party/obsmirror/LICENSE DESTINATION share/doc/dome-capture)
