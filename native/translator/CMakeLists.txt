cmake_minimum_required(VERSION 3.21)
project(dome-translator LANGUAGES CXX)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NATIVE REQUIRED IMPORTED_TARGET gio-2.0 fontconfig zlib)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include(GNUInstallDirs)
add_executable(dome-translator main.cpp notifier.cpp)
target_compile_features(dome-translator PRIVATE cxx_std_17)
target_compile_options(dome-translator PRIVATE -Wall -Wextra)
target_link_libraries(dome-translator PRIVATE fltk::fltk-shared fltk::images-shared PkgConfig::NATIVE)
target_compile_definitions(dome-translator PRIVATE TRANSLATOR_BUILD_DIR="${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(dome-translator PROPERTIES INSTALL_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>")
configure_file(backend.py backend.py COPYONLY)
install(TARGETS dome-translator RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES backend.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/dome-translator)
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/vendor/" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/dome-translator/vendor PATTERN __pycache__ EXCLUDE)
install(FILES ../../packaging/org.domed.Translator.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES translator.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME org.domed.Translator.svg)
include(CTest)
add_test(NAME translator-core COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/translator/test_backend.py)
set_tests_properties(translator-core PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/vendor")
if(BUILD_TESTING)
  add_executable(translator-source-fixture ../../tests/translator/source.cpp)
  target_link_libraries(translator-source-fixture PRIVATE fltk::fltk-shared)
  add_executable(translator-test-driver ../../tests/translator/driver.cpp notifier.cpp)
  target_compile_features(translator-test-driver PRIVATE cxx_std_17)
  target_compile_options(translator-test-driver PRIVATE -Wall -Wextra)
  target_link_libraries(translator-test-driver PRIVATE fltk::fltk-shared fltk::images-shared PkgConfig::NATIVE)
  target_compile_definitions(translator-test-driver PRIVATE
    TRANSLATOR_BUILD_DIR="${CMAKE_CURRENT_BINARY_DIR}"
    TRANSLATOR_TEST_WORKER="${CMAKE_CURRENT_SOURCE_DIR}/../../tests/translator/backend_fixture.py"
    TRANSLATOR_TEST_BINARY="$<TARGET_FILE:dome-translator>")
endif()
install(FILES ../../docs/TRANSLATOR.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/domed)
