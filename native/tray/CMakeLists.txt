cmake_minimum_required(VERSION 3.21)
project(dome-tray LANGUAGES CXX)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
find_package(Threads REQUIRED)
add_executable(dome-tray main.cpp services.cpp notifier.cpp)
foreach(size 24 48)
  set(icon "${CMAKE_CURRENT_SOURCE_DIR}/../icons/tray-applications-${size}.png")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${icon}")
  file(READ "${icon}" icon_hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," icon_bytes "${icon_hex}")
  set(TRAY_ICON_${size} "${icon_bytes}")
endforeach()
configure_file(tray-icon.h.in tray-icon.h @ONLY)
target_compile_features(dome-tray PRIVATE cxx_std_17)
target_compile_options(dome-tray PRIVATE -Wall -Wextra)
target_include_directories(dome-tray PRIVATE ..)
target_link_libraries(dome-tray PRIVATE fltk::fltk-shared fltk::gl-shared fltk::images-shared PkgConfig::FONTCONFIG Threads::Threads)
include(desktop-menu.cmake)
de_xr_desktop_menu(dome-tray)
set_target_properties(dome-tray PROPERTIES INSTALL_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>")
include(GNUInstallDirs)
install(TARGETS dome-tray RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
include(CTest)
if(BUILD_TESTING)
  add_executable(tray-service-driver ../../tests/tray/driver.cpp services.cpp)
  target_include_directories(tray-service-driver PRIVATE .)
  target_compile_features(tray-service-driver PRIVATE cxx_std_17)
  target_link_libraries(tray-service-driver PRIVATE PkgConfig::DESKTOPMENU)
endif()
